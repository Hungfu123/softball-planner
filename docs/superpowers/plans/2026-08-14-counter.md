# Umpire-Counter Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Eine Zähler-Seite unter `/counter` für Balls, Strikes, Outs, Inning und Score — ohne Login, ohne Netz, ohne Backend.

**Architecture:** Reine Frontend-Feature. Regel-Logik als pure Funktionen in `src/utils/counterRules.ts`, Zustand + Persistenz im Hook `useCounter` (localStorage, Undo als Snapshot-Stack), Darstellung in zwei Präsentationskomponenten `CounterBoard` (Anzeige) und `CounterPad` (Bedienung). **Tech Stack:** React 19, TypeScript, React Router DOM 7, Tailwind v4, Vitest + @testing-library/react.

**Spec:** `docs/superpowers/specs/2026-08-14-counter-design.md`

/**
 * Regel-Logik des Umpire-Counters — pure Funktionen, kein React, kein Storage.
 * Jede Funktion nimmt einen Zustand und gibt einen neuen zurück; nichts wird mutiert.
 * Undo im Hook arbeitet mit Snapshots, es gibt hier bewusst keine Umkehrfunktionen.
 */

export type Half = 'top' | 'bottom'

export interface CounterState {
  balls: number
  strikes: number
  outs: number
  inning: number
  half: Half
  home: number
  away: number
}

const MAX_BALLS = 4
const MAX_STRIKES = 3
const MAX_OUTS = 3

export const INITIAL_STATE: CounterState = {
  balls: 0,
  strikes: 0,
  outs: 0,
  inning: 1,
  half: 'top',
  home: 0,
  away: 0,
}

function clearCount(s: CounterState): CounterState {
  return { ...s, balls: 0, strikes: 0 }
}

/** Halbinning-Wechsel: alles auf 0, Seite kippt. Das Inning steigt nur von unten nach oben. */
export function advanceHalfInning(s: CounterState): CounterState {
  const toTop = s.half === 'bottom'
  return {
    ...s,
    balls: 0,
    strikes: 0,
    outs: 0,
    half: toTop ? 'top' : 'bottom',
    inning: toTop ? s.inning + 1 : s.inning,
  }
}

/** Der dritte Out wechselt sofort das Halbinning — die Tafel steht nie auf 3 Outs. */
export function addOut(s: CounterState): CounterState {
  if (s.outs + 1 >= MAX_OUTS) return advanceHalfInning(s)
  return clearCount({ ...s, outs: s.outs + 1 })
}

export function addStrike(s: CounterState): CounterState {
  if (s.strikes + 1 >= MAX_STRIKES) return addOut(s)
  return { ...s, strikes: s.strikes + 1 }
}

/** Foul zählt als Strike, aber nie zum dritten — bei 2 Strikes passiert nichts. */
export function addFoul(s: CounterState): CounterState {
  if (s.strikes >= MAX_STRIKES - 1) return s
  return { ...s, strikes: s.strikes + 1 }
}

/** Der vierte Ball ist ein Walk: Count zurück, kein Out. */
export function addBall(s: CounterState): CounterState {
  if (s.balls + 1 >= MAX_BALLS) return clearCount(s)
  return { ...s, balls: s.balls + 1 }
}

export function addRun(s: CounterState, team: 'home' | 'away'): CounterState {
  return team === 'home' ? { ...s, home: s.home + 1 } : { ...s, away: s.away + 1 }
}

/** Oben schlägt der Gast, unten das Heimteam. */
export function battingTeam(s: CounterState): 'home' | 'away' {
  return s.half === 'top' ? 'away' : 'home'
}

beforeEach(() => {

    expect(result.current.state.strikes).toBe(1)

    expect(result.current.state.strikes).toBe(2)
    act(() => result.current.undo())

```ts
import { useCallback, useMemo, useState } from 'react'
import {
  INITIAL_STATE,
  addBall,
  addFoul,
  addOut,
  addRun,
  addStrike,
  advanceHalfInning,
  type CounterState,
} from '@/utils/counterRules'

export const COUNTER_STORAGE_KEY = 'counter'

/**
 * Notbremse gegen unbegrenztes Wachstum. Ein Snapshot ist ~60 Byte, ein volles
 * Spiel bleibt weit darunter — der Cap greift praktisch nie.
 */
export const MAX_HISTORY = 1000

interface Stored {
  current: CounterState
  /** Zustand jeweils VOR einer Aktion, ältester zuerst. */
  history: CounterState[]
}

const EMPTY: Stored = { current: INITIAL_STATE, history: [] }

function isCounterState(value: unknown): value is CounterState {
  if (typeof value !== 'object' || value === null) return false
  const s = value as Record<string, unknown>
  return (
    typeof s.balls === 'number' &&
    typeof s.strikes === 'number' &&
    typeof s.outs === 'number' &&
    typeof s.inning === 'number' &&
    (s.half === 'top' || s.half === 'bottom') &&
    typeof s.home === 'number' &&
    typeof s.away === 'number'
  )
}

function load(): Stored {
  try {
    const raw = localStorage.getItem(COUNTER_STORAGE_KEY)
    if (!raw) return EMPTY
    const parsed: unknown = JSON.parse(raw)
    if (typeof parsed !== 'object' || parsed === null) return EMPTY
    const { current, history } = parsed as Record<string, unknown>
    if (!isCounterState(current)) return EMPTY
    return {
      current,
      history: Array.isArray(history) ? history.filter(isCounterState) : [],
    }
  } catch {
    return EMPTY
  }
}

function save(stored: Stored) {
  try {
    localStorage.setItem(COUNTER_STORAGE_KEY, JSON.stringify(stored))
  } catch {
    // Private Mode oder volles Kontingent — der Zähler läuft im Speicher weiter.
  }
}

/**
 * Zustand des Umpire-Counters mit localStorage-Persistenz.
 *
 * Undo ist ein Snapshot-Pop, kein Rückrechnen: vor jeder Aktion wird der alte
 * Zustand auf den Stack gelegt. Damit kann keine Umkehrlogik von der
 * Vorwärtslogik abweichen.
 */
export function useCounter() {
  const [stored, setStored] = useState<Stored>(load)

  const apply = useCallback((transition: (s: CounterState) => CounterState) => {
    setStored(prev => {
      const next: Stored = {
        current: transition(prev.current),
        history: [...prev.history, prev.current].slice(-MAX_HISTORY),
      }
      save(next)
      return next
    })
  }, [])

  const undo = useCallback(() => {
    setStored(prev => {
      if (prev.history.length === 0) return prev
      const next: Stored = {
        current: prev.history[prev.history.length - 1],
        history: prev.history.slice(0, -1),
      }
      save(next)
      return next
    })
  }, [])

  const reset = useCallback(() => {
    save(EMPTY)
    setStored(EMPTY)
  }, [])

  const actions = useMemo(
    () => ({
      strike: () => apply(addStrike),
      ball: () => apply(addBall),
      foul: () => apply(addFoul),
      out: () => apply(addOut),
      halfInning: () => apply(advanceHalfInning),
      runHome: () => apply(s => addRun(s, 'home')),
      runAway: () => apply(s => addRun(s, 'away')),
    }),
    [apply],
  )

  return {
    state: stored.current,
    canUndo: stored.history.length > 0,
    ...actions,
    undo,
    reset,
  }
}
```

interface Props {
  state: CounterState
  backTo: string
  backLabel: string
}

interface LampsProps {
  label: string
  count: number
  total: number
  /** Tailwind-Klasse für die leuchtende Lampe. */
  onClass: string
}

/**
 * Ball/Strike/Out als Lampenreihe statt Ziffer: drei Punkte erfasst man peripher,
 * eine Ziffer muss man lesen. Out und Strike haben nur zwei Lampen — beim dritten
 * greift sofort Out bzw. Halbinning-Wechsel, die Tafel steht nie auf dem Maximum.
 */
function Lamps({ label, count, total, onClass }: LampsProps) {
  return (
    <div className="flex-1 text-center">
      <div className="font-display text-[10px] font-bold uppercase tracking-[0.18em] text-navy-100/70">
        {label}
      </div>
      <div className="mt-1.5 flex justify-center gap-1.5" role="img" aria-label={`${label}: ${count} von ${total}`}>
        {Array.from({ length: total }, (_, i) => (
          <span
            key={i}
            className={`h-4 w-4 rounded-full ring-1 ring-inset ring-white/20 ${i < count ? onClass : 'bg-white/10'}`}
          />
        ))}
      </div>
    </div>
  )
}

interface ScoreProps {
  label: string
  runs: number
  atBat: boolean
}

import { createContext, useCallback, useContext, useMemo, useState } from 'react'
import type { ReactNode } from 'react'
import { setAuth, hasAuth, clearAuth } from '@/api/client'
import { LoginPage } from './LoginPage'

interface AuthValue {
  loggedIn: boolean
  error: string | null
  login: (password: string) => Promise<void>
}

const AuthContext = createContext<AuthValue | null>(null)

export function AuthProvider({ children }: { children: ReactNode }) {
  const [loggedIn, setLoggedIn] = useState(hasAuth())
  const [error, setError] = useState<string | null>(null)

  const login = useCallback(async (password: string) => {
    setAuth(password)
    try {
      const res = await fetch('/api/players', {
        headers: { Authorization: `Basic ${btoa(`admin:${password}`)}` },
      })
      if (res.status === 401) {
        clearAuth()
        setError('Falsches Passwort')
        return
      }
      setError(null)
      setLoggedIn(true)
    } catch {
      clearAuth()
      setError('Server nicht erreichbar')
    }
  }, [])

  const value = useMemo(() => ({ loggedIn, error, login }), [loggedIn, error, login])

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>
}

/**
 * Das Auth-Gate der App. Jede Route außer /counter liegt dahinter — genau eine
 * Stelle, an der geprüft wird, statt eines zweiten Pfads neben dem Router.
 */
export function RequireAuth({ children }: { children: ReactNode }) {
  const auth = useContext(AuthContext)
  if (!auth) throw new Error('RequireAuth braucht einen AuthProvider darüber')
  if (!auth.loggedIn) return <LoginPage onLogin={auth.login} error={auth.error} />
  return <>{children}</>
}
